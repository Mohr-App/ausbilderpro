rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null
        && (get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin'
         || get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'verwaltung');
    }

    // Ausbilder brauchen den gesamten Teilnehmer-Pool: TN suchen, Kurse besetzen,
    // Namenssuche, Dubletten finden. Deshalb eine eigene Rolle - vergeben wird sie
    // ausschliesslich von VerwaltungPro aus der Admin-Sitzung (siehe users-Regeln).
    function isAusbilder() {
      return request.auth != null
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'ausbilder';
    }

    // ─────────────────────────────────────────────────────────────
    // FIX 1 (KRITISCH): Rechteausweitung ueber das eigene role-Feld
    //
    // Vorher stand hier eine einzige Regel:
    //     allow read, write: if request.auth != null && request.auth.uid == uid;
    // Damit durfte jeder sein EIGENES User-Dokument komplett schreiben - auch das
    // Feld 'role'. Wer sich role:'admin' eintraegt, liefert ab dann in isAdmin()
    // ueberall true: Vollzugriff auf alle Teilnehmer, alle Kurse, den Audit-Log.
    //
    // Jetzt: Schreibrechte bleiben, aber 'role' ist fuer den Nutzer selbst tabu.
    // Rollen vergibt ausschliesslich die Verwaltung (isAdmin).
    // ─────────────────────────────────────────────────────────────
    match /users/{uid} {
      allow read: if request.auth != null && request.auth.uid == uid;
      allow read, write: if isAdmin();

      // Anlegen: Selbstregistrierung darf eine Rolle setzen - aber nur eine, die
      // KEINE zusaetzlichen Rechte verleiht. 'admin' und 'verwaltung' sind damit
      // ausgeschlossen: die Rechteausweitung ist zu.
      //
      // Diese Liste enthaelt exakt die Rollen, die die Apps selbst vergeben:
      //   arbeitgeber  - AG-Selbstregistrierung (ArbeitgeberPro)
      //   azubi        - AzubiPro-Registrierung
      //   arbeitnehmer - MitarbeiterPro (schreibt genau diesen Wert beim Login!)
      //
      // 'ausbilder' steht bewusst NICHT hier: die Rolle gibt Lesezugriff auf ALLE
      // Teilnehmer. Sie wird ausschliesslich von VerwaltungPro vergeben, und zwar
      // aus der Admin-Sitzung - da greift die isAdmin()-Regel oben.
      // AusbilderPro selbst hat keine Registrierung; beim Login legt es das Dokument
      // ohne role an, das ist hier ausdruecklich erlaubt (kein 'role' im Payload).
      allow create: if request.auth != null && request.auth.uid == uid
        && (!('role' in request.resource.data)
            || request.resource.data.role in ['arbeitgeber','azubi','arbeitnehmer']);

      // Aendern: alles ausser 'role'. Wer eine Rolle hat, behaelt sie; wer keine
      // hat, kann sich auch keine geben. Rollenwechsel macht nur die Verwaltung.
      //
      // WICHTIG: affectedKeys() listet nur GEAENDERTE Felder. Die Apps schreiben bei
      // jedem Login role mit (AzubiPro 'azubi', ArbeitgeberPro 'arbeitgeber',
      // MitarbeiterPro 'arbeitnehmer') - solange der Wert derselbe bleibt, ist das
      // keine Aenderung und laeuft durch. Nur ein echter Rollenwechsel wird geblockt.
      allow update: if request.auth != null && request.auth.uid == uid
        && !request.resource.data.diff(resource.data).affectedKeys().hasAny(['role']);

      // Fest einprogrammierte Ausnahme fuer den VerwaltungPro-Admin-Bypass (Stefan Mohr).
      // Die E-Mail steht direkt in der Regel (nicht in editierbaren Daten) - das ist sicher,
      // weil nur Firebase Auth selbst bestaetigt, wem dieses Token gehoert. Ohne diese
      // Zeile kann der App-Bypass zwar "ADMIN" anzeigen, aber Storage-Regeln (Kurs-
      // Dokumente etc.), die echte role='admin' pruefen, lehnen ihn trotzdem ab.
      allow write: if request.auth != null && request.auth.uid == uid
        && request.auth.token.email == 's.mohr@ish.de'
        && request.resource.data.role == 'admin';

      // Loeschen des eigenen Kontos bleibt erlaubt
      allow delete: if request.auth != null && request.auth.uid == uid;

      // Lookup fuer firmaCode-Verknuepfung: jeder authentifizierte User darf
      // Arbeitgeber-User-Docs lesen (enthaelt keine sensiblen Daten - nur firma/firmaCode/role/email)
      allow read: if request.auth != null
        && resource.data.role == 'arbeitgeber';
      // AG kann seine MA-User lesen (firma muss uebereinstimmen)
      allow read: if request.auth != null
        && resource.data.role == 'mitarbeiter';
      // Ausbilder duerfen Stammdaten anderer Ausbilder lesen (fuer Chat-Liste + Avatare)
      // HINWEIS: lastApp ist vom Nutzer selbst setzbar und daher KEIN Ausweis.
      // Fuer Lesezugriff auf Namen/Avatare ist das vertretbar - fuer Rechte NICHT.
      allow read: if request.auth != null
        && resource.data.lastApp == 'AusbilderPro';

      match /{sub}/{subDoc} {
        allow read, write: if request.auth != null && request.auth.uid == uid;
        allow read, write: if isAdmin();
        match /{deep}/{deepDoc} {
          allow read, write: if request.auth != null && request.auth.uid == uid;
          allow read, write: if isAdmin();
        }
      }
    }

          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['berufsschultage']);
        allow update: if request.auth != null
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['kurstage']);

          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['qrGeo']);
        // QR-Anwesenheit: GPS-Pflicht-Schalter
        allow update: if request.auth != null
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['qrGpsPflicht']);

        // dort als 'nicht zustaendig' abgelehnt, obwohl es sein Kurs ist.
        allow update: if request.auth != null
          && resource.data.ausbilderLoginEmail == request.auth.token.email
          && !('ausbilderUid' in resource.data)
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['ausbilderUid','ownerUid']);

        allow create: if request.auth != null
          && docId == 'kursSettings' && sub == 'ausbilder'
          && (subDoc == request.auth.token.email
              || request.resource.data.loginEmail == request.auth.token.email);
        allow update: if request.auth != null
          && docId == 'kursSettings' && sub == 'ausbilder'
          && (subDoc == request.auth.token.email
              || resource.data.loginEmail == request.auth.token.email
              || request.resource.data.loginEmail == request.auth.token.email);
        // Gemeinsame Nachschlage-Listen (Lehrgangsarten, Berufe, Raeume, Standorte):
        // jeder eingeloggte Ausbilder darf Eintraege ergaenzen/aktualisieren
        allow create, update: if request.auth != null
          && docId == 'kursSettings'
          && (sub == 'lehrgangsarten' || sub == 'berufe' || sub == 'raeume' || sub == 'standorte');
        match /{deep}/{deepDoc} {
          allow read: if request.auth != null;
          allow write: if isAdmin();
          // Selbstregistrierung: Azubi darf sich in Kurs eintragen
          allow create: if request.auth != null && deep == 'teilnehmer';

          // Ausbilder und Verwaltung: jeden TN
          allow delete: if (isAusbilder() || isAdmin()) && deep == 'teilnehmer';

          // und die App raeumt dabei seinen Kurs-Eintrag ab.
          // deepDoc ist die globalTN-ID; ueber verknuepftUid pruefen wir, dass es der
          // eigene Datensatz ist.
          allow delete: if request.auth != null && deep == 'teilnehmer'
            && get(/databases/$(database)/documents/globalTN/$(deepDoc)).data.verknuepftUid == request.auth.uid;
          // ─────────────────────────────────────────────────────────────
          // FIX 4 (KRITISCH): Noten, Anwesenheit, Bescheinigung
          //
          // Vorher: allow update: if request.auth != null && hasOnly([...]);
          // Die Regel prueft, WELCHE Felder geschrieben werden - aber nicht, WER
          // schreibt. Jeder eingeloggte Nutzer konnte bei JEDEM Teilnehmer Noten
          // aendern, Anwesenheit faelschen oder eine Bescheinigung ausstellen.
          //
          // Geschrieben wird das ausschliesslich von AusbilderPro und VerwaltungPro.
          // ArbeitgeberPro und MitarbeiterPro LESEN nur (0 Schreibzugriffe).
          // AzubiPro schreibt nur wunschSitzplatz, kenntnisAm und den Selbstbeitritt

          allow update: if (isAusbilder() || isAdmin())
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(

            );
          // Ausbilder-Freigabe des Teilnehmers - AzubiPro liest das nur (Z.4084)
          allow update: if (isAusbilder() || isAdmin())
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
              ['bestaetigtAm','bestaetigtVon']
            );
          // Azubi-Kenntnisnahme des Lehrgangs (eigenes Feld, NICHT die Ausbilder-Freigabe)
          // Kenntnisnahme ODER aktive Ablehnung ("Das ist nicht mein Lehrgang");
          // Bestaetigen loescht die Ablehnungs-Felder wieder (FieldValue.delete zaehlt als affectedKey)
          allow update: if request.auth != null
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
              ['kenntnisAm','kenntnisVon','kenntnisAbgelehntAm','kenntnisAbgelehntVon']
            );
          allow update: if request.auth != null
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['wunschSitzplatz']);
          // Merge-Operation: Ausbilder fuehrt manuellen + selbst-registrierten Eintrag
          // zusammen. mergedFromManual schreibt ausschliesslich AusbilderPro.
          allow update: if (isAusbilder() || isAdmin())
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
              ['firma','beruf','berufGruppeId','ausbjahr','kontaktEmail','kontaktTelefon','wunschSitzplatz','mergedFromManual']
            );
          // Azubi-Selbstbeitritt zu weiterem Kurs (processJoinUrl): markiert eigenen Eintrag
          allow update: if request.auth != null
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
              ['vorname','nachname','firma','beruf','ausbjahr','kontaktEmail','kontaktTelefon','status','addedAt','selfRegistered','bestaetigungAngefordert']
            );
          allow create, delete: if request.auth != null && deep == 'dokumente';
          // Meldungen: Azubi darf erstellen, Ausbilder darf lesen/loeschen
          allow create: if request.auth != null && deep == 'meldungen';
          allow delete: if request.auth != null && deep == 'meldungen';
          allow update: if request.auth != null && deep == 'meldungen'
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['gelesen','gelesenVon','gelesenAm']);
          // ─────────────────────────────────────────────────────────────
          // FIX 6: Anwesenheit fuer FREMDE melden
          //
          // Die alte Regel prueft, dass die Dokument-ID zum Inhalt passt (verhindert
          // Doppel-Check-in) - aber NICHT, ob die tnId dem Absender gehoert. Ein Azubi
          // konnte eine fremde tnId eintragen und jemanden anwesend melden, der gar
          // nicht da ist. Das landet ueber die Anwesenheit in der Bescheinigung, die

          //
          // Jetzt: Der Azubi nur fuer sich selbst. Ausbilder/Verwaltung fuer jeden -
          // das braucht tnFreigeben() (AusbilderPro Z.8285): Bei der Freigabe wird der
          // Azubi gleich anwesend gesetzt, weil er bis dahin nicht einchecken konnte.
          // ─────────────────────────────────────────────────────────────
          allow create: if request.auth != null && deep == 'checkins'
            && request.resource.data.keys().hasAll(['tnId','datum','zeit'])
            && deepDoc == request.resource.data.tnId + '_' + request.resource.data.datum
            && (isAusbilder() || isAdmin()
                || get(/databases/$(database)/documents/globalTN/$(request.resource.data.tnId)).data.verknuepftUid == request.auth.uid);
          // tnFreigeben() schreibt mit merge:true - das ist ein Update, wenn der
          // Eintrag schon existiert. Deshalb hier ebenfalls erlaubt, aber nur fuer
          // Ausbilder/Verwaltung: der Azubi darf einen Check-in nicht nachtraeglich
          // veraendern.
          allow update: if (isAusbilder() || isAdmin()) && deep == 'checkins';
          // Aufraeumen: Ausbilder/Verwaltung (AusbilderPro Z.6367 beim Sitzplatzwechsel)
          allow delete: if (isAusbilder() || isAdmin()) && deep == 'checkins';
          // ─────────────────────────────────────────────────────────────
          // v847 Live-Sync (Beta): Sitzplan + Tagesanwesenheit als Live-
          // Dokumente am Kurs (live/sitzplan, live/anw_<datum>). Nur
          // Ausbilder/Verwaltung - Azubis lesen und schreiben hier nicht.
          // ─────────────────────────────────────────────────────────────
          allow read, create, update, delete: if (isAusbilder() || isAdmin()) && deep == 'live';
        }
      }
    }

        && resource.data.verknuepftUid != null
        && (!('verknuepftFreigegebenAm' in resource.data) || resource.data.verknuepftFreigegebenAm == null)
        && request.resource.data.verknuepftUid == null;
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'wiederholer']);
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['photoUrl']);
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['kontaktEmail','kontaktTelefon','wunschSitzplatz','dsgvoZustimmung','arbeitszeitPlan','materialListe','kundenListe','geofence','geburtsdatum']);
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['urlaub']);
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
          ['benachrichtigungen','aktiverKursId','pendingRemoval','pendingRemovalKurs','kursHistory','wunschSitzplatz','status','wiederholer']
        );
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['beruf']);
      // Merge-Operation (Ausbilder fuehrt manuell + selbstregistriert zusammen)
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
          ['firma','beruf','berufGruppeId','ausbjahr','kontaktEmail','kontaktTelefon','wunschSitzplatz','aktiverKursId','kursHistory','mergedFromManual']
        );
      // Firmencode-Verknuepfung: Azubi verknuepft sich mit Arbeitgeber
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['firmaUid','firma','arbeitgeberId','betriebId']);
      // Firmencode-Verknuepfungs-Antrag (pending) - Azubi setzt es bei Code-Eingabe
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['firmaPending']);
      // AG genehmigt: setzt firmaUid/firma + loescht firmaPending + setzt Verknuepfungs-Beginn
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['firmaUid','firma','arbeitgeberId','betriebId','firmaPending','_verknuepftSeit']);
      // Trennen: Verknuepfungs-Felder loeschen (Daten bleiben fuer Aufbewahrungspflicht).
      // 'firma' bleibt als Kammerdatum erhalten, firmaGetrenntVon sperrt die
      // automatische Wieder-Zuordnung aus der KH-Liste.
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['firmaUid','firma','arbeitgeberId','betriebId','_verknuepftSeit','firmaGetrenntVon']);
      // Automatische Zuordnung aus der KH-Liste (VerwaltungPro)
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['firmaUid','_verknuepftSeit','_verknuepftVon']);
      // AG darf Stammdaten seiner verknuepften MAs editieren (Vorname/Nachname/Geb./Adresse/Tel/Email/Beruf/Lehrjahr/Foto)
      allow update: if request.auth != null
        && resource.data.firmaUid == request.auth.uid
        && request.resource.data.firmaUid == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
          ['vorname','nachname','geburtsdatum','kontaktTelefon','kontaktEmail','adresse','plz','ort','beruf','ausbjahr','photoUrl']
        );
      // Ausbilder darf Vorname/Nachname korrigieren (z.B. nach Pilot-Modus-Beendigung
      // wenn Klarnamen wiederhergestellt werden muessen)
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['vorname','nachname']);
      // Pilot-Anonymisierung: Admin darf Datenminimierungs-Felder updaten
      allow update: if isAdmin()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
          ['vorname','nachname','tel','kontaktTelefon','geburtsdatum','adresse','plz','ort','pilotAnonymizedAt']
        );
      match /dokumente/{fileDoc} {
        allow read: if request.auth != null;
        allow create, delete: if request.auth != null;
      }
      match /zeiterfassung/{dayDoc} {
        allow read, write: if request.auth != null;
      }
      match /urlaubsantraege/{antragDoc} {
        allow read, write: if request.auth != null;
      }
      match /baustellen/{bsDoc} {
        allow read, write: if request.auth != null;
      }
      match /auftraege/{aufDoc} {
        allow read, write: if request.auth != null;
      }
      match /betriebMeldungen/{meldDoc} {
        allow read, write: if request.auth != null;
      }
      // Firma-Historie - speichert beendete Verknuepfungen mit Periode
      match /firmaHistorie/{histDoc} {
        allow read, write: if request.auth != null;
      }
    }

    match /mail/{mailId} {
      allow create: if request.auth != null;
      allow read, update, delete: if false; // Nur Extension
    }

    // Verwaltung-Events (Aufploppen-Notifications fuer VerwaltungPro)
    // - Ausbilder erstellt Events bei kritischen Status-Aenderungen (absent, wiederholer)
    // - Verwaltung liest und quittiert (Betrieb informiert)
    match /vwEvents/{eventId} {
      allow create: if request.auth != null
        && request.resource.data.keys().hasAll(['typ','tnId','tnName','erstellt','gelesen'])
        && request.resource.data.typ in ['absent','wiederholer']
        && request.resource.data.gelesen == false;
      allow read: if request.auth != null;
      // Verwaltung darf gelesen + Quittier-Felder aendern
      // Ausbilder darf direkt nach Anlage Mail-Status-Felder nachtragen (async-Check)
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
          ['gelesen','quittiert','quittiertAm','quittiertVon','quittiertNotiz',
           'agMailGesendet','agMailGrund','agAnzahlEmpfaenger']
        );
      // Loeschen erlaubt - alte Events koennen aufgeraeumt werden
      allow delete: if request.auth != null;
    }

    // collectionGroup-Query auf firmaHistorie: AG findet alle Subcollection-Eintraege wo seine firmaUid drinsteht
    match /{path=**}/firmaHistorie/{histDoc} {
      allow read: if request.auth != null;
    }

    // collectionGroup-Query auf berichtsheft (Phase 2): AG liest die abgegebenen
    // Wochenberichte seiner verknuepften Azubis (firmaUid == seine UID) und darf
    // ausschliesslich Bestaetigungs-/Status-Felder schreiben. Der Azubi (Besitzer)
    // hat Voll-Zugriff bereits ueber die generische users/{uid}/{sub}-Regel.
    match /{path=**}/berichtsheft/{w} {
      allow read: if request.auth != null && resource.data.firmaUid == request.auth.uid;
      allow update: if request.auth != null
        && resource.data.firmaUid == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys()
           .hasOnly(['status','agSignatur','agSignDatum','agKommentar','geaendertAt']);
    }

    // Audit-Log: nur Admin darf schreiben, niemand darf modifizieren/loeschen
    match /auditLog/{logId} {
      allow read: if isAdmin();
      allow create: if isAdmin();
      allow update, delete: if false;
    }

    // ═══════════════════════════════════════════════════════════
    // Ausbilder-Kommunikation: 1:1 + Gruppen-Chats
    // ═══════════════════════════════════════════════════════════
    match /chats/{chatId} {
      // Lesen: nur Teilnehmer
      allow read: if request.auth != null
        && request.auth.uid in resource.data.participants;

      // Erstellen: jeder authentifizierte User, Ersteller muss in participants sein
      allow create: if request.auth != null
        && request.auth.uid in request.resource.data.participants
        && request.resource.data.createdBy == request.auth.uid;

      // Update 1: Nachrichten-Metadaten (lastMessage/unread) - jeder Teilnehmer
      allow update: if request.auth != null
        && request.auth.uid in resource.data.participants
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['lastMessage','unread']);

      // Update 2: Gruppenname aendern - nur Admin
      allow update: if request.auth != null
        && resource.data.type == 'group'
        && resource.data.admin == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['name']);

      // Update 3: Mitglied hinzufuegen/entfernen - nur Admin (in Gruppen)
      allow update: if request.auth != null
        && resource.data.type == 'group'
        && resource.data.admin == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['participants','joinedAt','unread']);

      // Update 4: Selbst verlassen (Gruppen) - inkl. Admin-Uebergabe
      allow update: if request.auth != null
        && resource.data.type == 'group'
        && request.auth.uid in resource.data.participants
        && !(request.auth.uid in request.resource.data.participants)
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['participants','joinedAt','admin','unread']);

      // Loeschen: nur Admin einer leeren Gruppe (Aufraeum-Fall)
      allow delete: if request.auth != null
        && resource.data.createdBy == request.auth.uid
        && resource.data.participants.size() <= 1;

      // Subcollection: Nachrichten
      match /messages/{msgId} {
        // Lesen: nur Teilnehmer des Chats
        allow read: if request.auth != null
          && request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.participants;

        // Erstellen: nur Teilnehmer + von muss eigene uid sein (System-Messages auch erlaubt)
        allow create: if request.auth != null
          && request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.participants
          && (request.resource.data.von == request.auth.uid
              || request.resource.data.system == true);

        // Update: nur eigene Nachrichten + nur Soft-Delete-Felder (inkl. attachment-Entfernung)
        allow update: if request.auth != null
          && resource.data.von == request.auth.uid
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['deleted','deletedAt','text','attachment']);

        // Hard-Delete: eigene Nachrichten, ODER Admin bei Nachrichten >60 Tage alt (Admin-Sweep)
        allow delete: if request.auth != null
          && (resource.data.von == request.auth.uid
              || (isAdmin()
                  && resource.data.ts is timestamp
                  && resource.data.ts.toMillis() < (request.time.toMillis() - 60 * 24 * 60 * 60 * 1000)));
      }
    }

    // ═══════════════════════════════════════════════════════════
    // Online-Presence (Wer ist online jetzt)
    // ═══════════════════════════════════════════════════════════
    match /presence/{uid} {
      // Jeder eingeloggte User darf alle Presence-Stati sehen
      allow read: if request.auth != null;
      // Jeder darf nur sein eigenes Presence-Doc schreiben (create/update/delete)
      allow write: if request.auth != null && request.auth.uid == uid;
    }

    // Warteschlange fuer die kombinierte Willkommens-/Bestaetigungsmail (Cloud Function sendWelcomeMail).
    // Azubi darf nur einen Auftrag fuer sich selbst anlegen; die Cloud Function (Admin) verarbeitet + loescht ihn.
    match /welcomeQueue/{id} {
      allow create: if request.auth != null
                    && request.resource.data.uid == request.auth.uid;
      allow read, delete: if request.auth != null
                    && resource.data.uid == request.auth.uid;
      allow update: if false;
    }
  }
