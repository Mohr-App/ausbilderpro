rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null
        && (get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin'
         || get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'verwaltung');
    }

    match /users/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
      allow read, write: if isAdmin();
      // Lookup für firmaCode-Verknüpfung: jeder authentifizierte User darf
      // Arbeitgeber-User-Docs lesen (enthält keine sensiblen Daten - nur firma/firmaCode/role/email)
      allow read: if request.auth != null
        && resource.data.role == 'arbeitgeber';
      // AG kann seine MA-User lesen (firma muss übereinstimmen)
      allow read: if request.auth != null
        && resource.data.role == 'mitarbeiter';
      // Ausbilder dürfen Stammdaten anderer Ausbilder lesen (für Chat-Liste + Avatare)
      allow read: if request.auth != null
        && resource.data.lastApp == 'AusbilderPro';
      // Profilbild-URL darf vom eigenen User aktualisiert werden
      allow update: if request.auth != null
        && request.auth.uid == uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['photoUrl']);
      match /{sub}/{subDoc} {
        allow read, write: if request.auth != null && request.auth.uid == uid;
        allow read, write: if isAdmin();
        match /{deep}/{deepDoc} {
          allow read, write: if request.auth != null && request.auth.uid == uid;
          allow read, write: if isAdmin();
        }
      }
    }

    match /shared/{docId} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
      match /{sub}/{subDoc} {
        allow read: if request.auth != null;
        allow write: if isAdmin();
        // Standalone: Ausbilder darf Kurse erstellen
        allow create: if request.auth != null && docId == 'kurse' && sub == 'list';
        // Standalone: Ausbilder darf eigene Kurse löschen
        allow delete: if request.auth != null && docId == 'kurse' && sub == 'list'
          && resource.data.createdBy == request.auth.token.email;
        allow update: if request.auth != null
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['urlaub']);
        allow update: if request.auth != null
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['benachrichtigungen']);
        allow update: if request.auth != null
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['berufsschultage']);
        allow update: if request.auth != null
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['kurstage']);
        // QR-Anwesenheit: Ausbilder setzt Kurs-Standort + Radius (Geofence)
        allow update: if request.auth != null
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['qrGeo']);
        // Profil-Backfill: Kurs-Ersteller darf eigene Ausbilder-Stammdaten nachtragen
        allow update: if request.auth != null
          && resource.data.createdBy == request.auth.token.email
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['ausbilderName','ausbilderEmail','ausbilderTel']);
        allow create: if request.auth != null && docId == 'urlaubLog';
        match /{deep}/{deepDoc} {
          allow read: if request.auth != null;
          allow write: if isAdmin();
          // Selbstregistrierung: Azubi darf sich in Kurs eintragen
          allow create: if request.auth != null && deep == 'teilnehmer';
          // Ausbilder darf TN aus Kurs löschen
          allow delete: if request.auth != null && deep == 'teilnehmer';
          allow update: if request.auth != null
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
              ['status','attTotal','attPresent','attAbsentE','attAbsentU','attSick','attBerufsschule','attLate','attEarly','attPercent','attLastSync','noten_mitarbeit','noten_fuehrung','noten_praxis','noten_extra','werkzeugStatus','werkzeugFehlt','bemerkung','bemerkungVon','bescheinigungSignatur','bescheinigungDatum','bescheinigungVon','bescheinigungAusgestellt']
            );
          allow update: if request.auth != null
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
              ['bestaetigtAm','bestaetigtVon']
            );
          allow update: if request.auth != null
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['wunschSitzplatz']);
          // Merge-Operation: Ausbilder fuehrt manuellen + selbst-registrierten Eintrag zusammen
          allow update: if request.auth != null
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
              ['firma','beruf','berufGruppeId','ausbjahr','kontaktEmail','kontaktTelefon','wunschSitzplatz','mergedFromManual']
            );
          allow create, delete: if request.auth != null && deep == 'dokumente';
          // Meldungen: Azubi darf erstellen, Ausbilder darf lesen/löschen
          allow create: if request.auth != null && deep == 'meldungen';
          allow delete: if request.auth != null && deep == 'meldungen';
          allow update: if request.auth != null && deep == 'meldungen'
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['gelesen','gelesenVon','gelesenAm']);
          // QR-Anwesenheit Check-ins: Azubi erstellt (nur fuer sich, heute, einmal - DocId tnId_datum
          // verhindert Doppel-Check-in), kein Update (kein Manipulieren), Ausbilder darf aufraeumen
          allow create: if request.auth != null && deep == 'checkins'
            && request.resource.data.keys().hasAll(['tnId','datum','zeit'])
            && deepDoc == request.resource.data.tnId + '_' + request.resource.data.datum;
          allow delete: if request.auth != null && deep == 'checkins';
        }
      }
    }

    match /globalTN/{doc} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
      allow delete: if request.auth != null;
      // Selbstregistrierung + manuelles Anlegen
      allow create: if request.auth != null;
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'wiederholer']);
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['photoUrl']);
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['kontaktEmail','kontaktTelefon','wunschSitzplatz','dsgvoZustimmung','arbeitszeitPlan','materialListe','kundenListe','geofence','geburtsdatum']);
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['urlaub']);
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
          ['benachrichtigungen','aktiverKursId','pendingRemoval','pendingRemovalKurs','kursHistory','wunschSitzplatz','status','wiederholer']
        );
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['beruf']);
      // Merge-Operation (Ausbilder fuehrt manuell + selbstregistriert zusammen)
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
          ['firma','beruf','berufGruppeId','ausbjahr','kontaktEmail','kontaktTelefon','wunschSitzplatz','aktiverKursId','kursHistory','mergedFromManual']
        );
      // Firmencode-Verknüpfung: Azubi verknüpft sich mit Arbeitgeber
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['firmaUid','firma','arbeitgeberId','betriebId']);
      // Firmencode-Verknuepfungs-Antrag (pending) - Azubi setzt es bei Code-Eingabe
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['firmaPending']);
      // AG genehmigt: setzt firmaUid/firma + loescht firmaPending + setzt Verknuepfungs-Beginn
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['firmaUid','firma','arbeitgeberId','betriebId','firmaPending','_verknuepftSeit']);
      // Trennen: Verknuepfungs-Felder loeschen (Daten bleiben fuer Aufbewahrungspflicht)
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['firmaUid','firma','arbeitgeberId','betriebId','_verknuepftSeit']);
      // AG darf Stammdaten seiner verknuepften MAs editieren (Vorname/Nachname/Geb./Adresse/Tel/Email/Beruf/Lehrjahr/Foto)
      allow update: if request.auth != null
        && resource.data.firmaUid == request.auth.uid
        && request.resource.data.firmaUid == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
          ['vorname','nachname','geburtsdatum','kontaktTelefon','kontaktEmail','adresse','plz','ort','beruf','ausbjahr','photoUrl']
        );
      // Ausbilder darf Vorname/Nachname korrigieren (z.B. nach Pilot-Modus-Beendigung
      // wenn Klarnamen wiederhergestellt werden muessen)
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['vorname','nachname']);
      // Pilot-Anonymisierung: Admin darf Datenminimierungs-Felder updaten
      allow update: if isAdmin()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
          ['vorname','nachname','tel','kontaktTelefon','geburtsdatum','adresse','plz','ort','pilotAnonymizedAt']
        );
      match /dokumente/{fileDoc} {
        allow read: if request.auth != null;
        allow create, delete: if request.auth != null;
      }
      match /zeiterfassung/{dayDoc} {
        allow read, write: if request.auth != null;
      }
      match /urlaubsantraege/{antragDoc} {
        allow read, write: if request.auth != null;
      }
      match /baustellen/{bsDoc} {
        allow read, write: if request.auth != null;
      }
      match /auftraege/{aufDoc} {
        allow read, write: if request.auth != null;
      }
      match /betriebMeldungen/{meldDoc} {
        allow read, write: if request.auth != null;
      }
      // Firma-Historie - speichert beendete Verknuepfungen mit Periode
      match /firmaHistorie/{histDoc} {
        allow read, write: if request.auth != null;
      }
    }

    match /mail/{mailId} {
      allow create: if request.auth != null;
      allow read, update, delete: if false; // Nur Extension
    }

    // Verwaltung-Events (Aufploppen-Notifications fuer VerwaltungPro)
    // - Ausbilder erstellt Events bei kritischen Status-Aenderungen (absent, wiederholer)
    // - Verwaltung liest und quittiert (Betrieb informiert)
    match /vwEvents/{eventId} {
      allow create: if request.auth != null
        && request.resource.data.keys().hasAll(['typ','tnId','tnName','erstellt','gelesen'])
        && request.resource.data.typ in ['absent','wiederholer']
        && request.resource.data.gelesen == false;
      allow read: if request.auth != null;
      // Verwaltung darf gelesen + Quittier-Felder aendern
      // Ausbilder darf direkt nach Anlage Mail-Status-Felder nachtragen (async-Check)
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
          ['gelesen','quittiert','quittiertAm','quittiertVon','quittiertNotiz',
           'agMailGesendet','agMailGrund','agAnzahlEmpfaenger']
        );
      // Loeschen erlaubt - alte Events koennen aufgeraeumt werden
      allow delete: if request.auth != null;
    }

    // collectionGroup-Query auf firmaHistorie: AG findet alle Subcollection-Eintraege wo seine firmaUid drinsteht
    match /{path=**}/firmaHistorie/{histDoc} {
      allow read: if request.auth != null;
    }

    // collectionGroup-Query auf berichtsheft (Phase 2): AG liest die abgegebenen
    // Wochenberichte seiner verknuepften Azubis (firmaUid == seine UID) und darf
    // ausschliesslich Bestaetigungs-/Status-Felder schreiben. Der Azubi (Besitzer)
    // hat Voll-Zugriff bereits ueber die generische users/{uid}/{sub}-Regel.
    match /{path=**}/berichtsheft/{w} {
      allow read: if request.auth != null && resource.data.firmaUid == request.auth.uid;
      allow update: if request.auth != null
        && resource.data.firmaUid == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys()
           .hasOnly(['status','agSignatur','agSignDatum','agKommentar','geaendertAt']);
    }

    // Audit-Log: nur Admin darf schreiben, niemand darf modifizieren/loeschen
    match /auditLog/{logId} {
      allow read: if isAdmin();
      allow create: if isAdmin();
      allow update, delete: if false;
    }

    // ═══════════════════════════════════════════════════════════
    // Ausbilder-Kommunikation: 1:1 + Gruppen-Chats
    // ═══════════════════════════════════════════════════════════
    match /chats/{chatId} {
      // Lesen: nur Teilnehmer
      allow read: if request.auth != null
        && request.auth.uid in resource.data.participants;

      // Erstellen: jeder authentifizierte User, Ersteller muss in participants sein
      allow create: if request.auth != null
        && request.auth.uid in request.resource.data.participants
        && request.resource.data.createdBy == request.auth.uid;

      // Update 1: Nachrichten-Metadaten (lastMessage/unread) – jeder Teilnehmer
      allow update: if request.auth != null
        && request.auth.uid in resource.data.participants
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['lastMessage','unread']);

      // Update 2: Gruppenname ändern – nur Admin
      allow update: if request.auth != null
        && resource.data.type == 'group'
        && resource.data.admin == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['name']);

      // Update 3: Mitglied hinzufügen/entfernen – nur Admin (in Gruppen)
      allow update: if request.auth != null
        && resource.data.type == 'group'
        && resource.data.admin == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['participants','joinedAt','unread']);

      // Update 4: Selbst verlassen (Gruppen) – inkl. Admin-Übergabe
      allow update: if request.auth != null
        && resource.data.type == 'group'
        && request.auth.uid in resource.data.participants
        && !(request.auth.uid in request.resource.data.participants)
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['participants','joinedAt','admin','unread']);

      // Löschen: nur Admin einer leeren Gruppe (Aufräum-Fall)
      allow delete: if request.auth != null
        && resource.data.createdBy == request.auth.uid
        && resource.data.participants.size() <= 1;

      // Subcollection: Nachrichten
      match /messages/{msgId} {
        // Lesen: nur Teilnehmer des Chats
        allow read: if request.auth != null
          && request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.participants;

        // Erstellen: nur Teilnehmer + von muss eigene uid sein (System-Messages auch erlaubt)
        allow create: if request.auth != null
          && request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.participants
          && (request.resource.data.von == request.auth.uid
              || request.resource.data.system == true);

        // Update: nur eigene Nachrichten + nur Soft-Delete-Felder (inkl. attachment-Entfernung)
        allow update: if request.auth != null
          && resource.data.von == request.auth.uid
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['deleted','deletedAt','text','attachment']);

        // Hard-Delete: eigene Nachrichten, ODER Admin bei Nachrichten >60 Tage alt (Admin-Sweep)
        allow delete: if request.auth != null
          && (resource.data.von == request.auth.uid
              || (isAdmin()
                  && resource.data.ts is timestamp
                  && resource.data.ts.toMillis() < (request.time.toMillis() - 60 * 24 * 60 * 60 * 1000)));
      }
    }

    // ═══════════════════════════════════════════════════════════
    // Online-Presence (Wer ist online jetzt)
    // ═══════════════════════════════════════════════════════════
    match /presence/{uid} {
      // Jeder eingeloggte User darf alle Presence-Stati sehen
      allow read: if request.auth != null;
      // Jeder darf nur sein eigenes Presence-Doc schreiben (create/update/delete)
      allow write: if request.auth != null && request.auth.uid == uid;
    }
  }
}
