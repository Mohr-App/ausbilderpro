rules_version = '2';

service firebase.storage {
  match /b/{bucket}/o {

    // ═══════════════════════════════════════════════════════════
    // Helfer-Funktionen (Rollen / Eigentümer-Prüfung via Firestore)
    // ═══════════════════════════════════════════════════════════
    function userRole() {
      return firestore.get(/databases/(default)/documents/users/$(request.auth.uid)).data.role;
    }
    function isVerwaltung() {
      return request.auth != null && (userRole() == 'admin' || userRole() == 'verwaltung');
    }
    function isAusbilder() {
      return request.auth != null && userRole() == 'ausbilder';
    }
    // Verwaltung ODER der Ausbilder, dem genau dieser Kurs gehört
    function isCourseStaff(kursId) {
      return isVerwaltung()
        || (isAusbilder()
            && (firestore.get(/databases/(default)/documents/shared/kurse/list/$(kursId)).data.ausbilderUid == request.auth.uid
             || firestore.get(/databases/(default)/documents/shared/kurse/list/$(kursId)).data.ownerUid == request.auth.uid));
    }
    // Der Azubi, der genau diesem TN-Datensatz zugeordnet ist
    function isThisTN(tnId) {
      return request.auth != null
        && firestore.get(/databases/(default)/documents/globalTN/$(tnId)).data.verknuepftUid == request.auth.uid;
    }
    // Erlaubte Office-/Dokument-Typen (KEINE Bilder)
    function isOfficeType() {
      return request.resource.contentType.matches('text/.*')
        || request.resource.contentType == 'application/pdf'
        || request.resource.contentType == 'application/msword'
        || request.resource.contentType == 'application/vnd.openxmlformats-officedocument.wordprocessingml.document'
        || request.resource.contentType == 'application/vnd.ms-excel'
        || request.resource.contentType == 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'
        || request.resource.contentType == 'application/vnd.ms-powerpoint'
        || request.resource.contentType == 'application/vnd.openxmlformats-officedocument.presentationml.presentation'
        || request.resource.contentType == 'application/octet-stream';
    }

    // ═══════════════════════════════════════════════════════════
    // Chat-Anhänge: Bilder und Dateien max. 1 MB
    // Path: chats/{chatId}/{msgId}/{filename}
    // ═══════════════════════════════════════════════════════════
    match /chats/{chatId}/{msgId}/{filename} {
      allow read: if request.auth != null
        && request.auth.uid in firestore.get(/databases/(default)/documents/chats/$(chatId)).data.participants;
      allow write: if request.auth != null
        && request.auth.uid in firestore.get(/databases/(default)/documents/chats/$(chatId)).data.participants
        && request.resource.size < 1 * 1024 * 1024
        && (request.resource.contentType.matches('image/.*')
            || request.resource.contentType == 'application/pdf'
            || request.resource.contentType.matches('text/.*')
            || request.resource.contentType == 'application/msword'
            || request.resource.contentType == 'application/vnd.openxmlformats-officedocument.wordprocessingml.document'
            || request.resource.contentType == 'application/vnd.ms-excel'
            || request.resource.contentType == 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'
            || request.resource.contentType == 'application/octet-stream');
      allow delete: if request.auth != null
        && request.auth.uid in firestore.get(/databases/(default)/documents/chats/$(chatId)).data.participants;
    }

    // ═══════════════════════════════════════════════════════════
    // Profilbilder: userProfiles/{uid}/avatar.jpg max 500 KB
    // ═══════════════════════════════════════════════════════════
    match /userProfiles/{uid}/{filename} {
      allow read: if request.auth != null;
      allow write: if request.auth != null
        && request.auth.uid == uid
        && request.resource.size < 500 * 1024
        && request.resource.contentType.matches('image/.*');
      allow delete: if request.auth != null
        && request.auth.uid == uid;
    }

    // ═══════════════════════════════════════════════════════════
    // Kursunterlagen – INTERN (nur Verwaltung + zustaendiger Ausbilder)
    // Path: shared/kurse/{kursId}/docs/{filename}  – max 20 MB
    // TN haben KEINEN Zugriff.
    // ═══════════════════════════════════════════════════════════
    match /shared/kurse/{kursId}/docs/{filename} {
      allow read:   if isCourseStaff(kursId);
      allow write:  if isCourseStaff(kursId)
        && request.resource.size < 20 * 1024 * 1024
        && (isOfficeType() || request.resource.contentType.matches('image/.*'));
      allow delete: if isCourseStaff(kursId);
    }

    // ═══════════════════════════════════════════════════════════
    // TN-Dokumente – Verwaltung + Ausbilder + der betroffene TN selbst
    // Path: shared/tn/{tnId}/docs/{filename}  – max 20 MB
    // NUR PDF / Word / Excel / PowerPoint / Text (KEINE Bilder).
    // Andere Azubis / Betriebe haben KEINEN Zugriff.
    // ═══════════════════════════════════════════════════════════
    match /shared/tn/{tnId}/docs/{filename} {
      allow read:   if isVerwaltung() || isAusbilder() || isThisTN(tnId);
      allow write:  if (isVerwaltung() || isAusbilder() || isThisTN(tnId))
        && request.resource.size < 20 * 1024 * 1024
        && isOfficeType();
      allow delete: if isVerwaltung() || isAusbilder() || isThisTN(tnId);
    }
  }
}
