rules_version = '2';

service firebase.storage {
  match /b/{bucket}/o {

    // ═══════════════════════════════════════════════════════════
    // Chat-Anhänge: Bilder und Dateien max. 1 MB
    // Path: chats/{chatId}/{msgId}/{filename}
    // ═══════════════════════════════════════════════════════════
    match /chats/{chatId}/{msgId}/{filename} {

      // Lesen: nur Teilnehmer des Chats
      allow read: if request.auth != null
        && request.auth.uid in firestore.get(/databases/(default)/documents/chats/$(chatId)).data.participants;

      // Hochladen: Teilnehmer + max 1 MB + nur Bilder/PDFs/Office/Text
      allow write: if request.auth != null
        && request.auth.uid in firestore.get(/databases/(default)/documents/chats/$(chatId)).data.participants
        && request.resource.size < 1 * 1024 * 1024
        && (request.resource.contentType.matches('image/.*')
            || request.resource.contentType == 'application/pdf'
            || request.resource.contentType.matches('text/.*')
            || request.resource.contentType == 'application/msword'
            || request.resource.contentType == 'application/vnd.openxmlformats-officedocument.wordprocessingml.document'
            || request.resource.contentType == 'application/vnd.ms-excel'
            || request.resource.contentType == 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'
            || request.resource.contentType == 'application/octet-stream');

      // Löschen: jeder Teilnehmer (eigene Files via App, fremde via Sweep – wird durch
      // Firestore-Rule auf Message kontrolliert: nur wer Message löscht, löscht auch das File)
      allow delete: if request.auth != null
        && request.auth.uid in firestore.get(/databases/(default)/documents/chats/$(chatId)).data.participants;
    }
  }
}
